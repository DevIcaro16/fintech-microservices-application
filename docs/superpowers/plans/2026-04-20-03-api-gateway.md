# API Gateway Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implementar o `api-gateway` em Bun + Elysia com pipeline de middlewares — rate limiting, autenticação via `auth-service`, propagação de `trace_id`, roteamento reverso para os serviços internos e endpoint `/metrics` para Prometheus.

**Architecture:** Pipeline de middlewares Elysia processados em ordem: logger → trace_id → rate limiter → auth guard → proxy. Cada request recebe um `trace_id` (UUID) que é propagado como header `X-Trace-ID` para todos os upstreams. O rate limiter usa Map in-memory com sliding window por IP. A autenticação chama `GET /auth/validate` no `auth-service` interno — sem verificação local de JWT. Rotas públicas (`/auth/login`, `/auth/refresh`, `/healthz`) bypassam o auth guard.

**Tech Stack:** Bun ≥1.1, Elysia ≥1.1, TypeScript, Docker (oven/bun), Kubernetes (namespace `fintech`).

// api-gateway/src/middleware/trace.ts
import { Elysia } from "elysia";

function generateTraceId(): string {
  const bytes = crypto.getRandomValues(new Uint8Array(16));
  bytes[6] = (bytes[6] & 0x0f) | 0x40;
  bytes[8] = (bytes[8] & 0x3f) | 0x80;
  const hex = Array.from(bytes, (b) => b.toString(16).padStart(2, "0")).join("");
  return `${hex.slice(0, 8)}-${hex.slice(8, 12)}-${hex.slice(12, 16)}-${hex.slice(16, 20)}-${hex.slice(20)}`;
}

export const tracePlugin = new Elysia({ name: "trace" }).derive(
  { as: "global" },
  ({ request }) => {
    const traceId =
      request.headers.get("x-trace-id") ?? generateTraceId();
    return { traceId };
  }
);
```

- [ ] **Step 2: Criar routes.ts**

// api-gateway/src/config/routes.ts
const AUTH_SERVICE =
  process.env.AUTH_SERVICE_URL ?? "http://auth-service.fintech.svc.cluster.local";

export const PUBLIC_ROUTES = new Set([
  "POST /auth/login",
  "POST /auth/refresh",
]);

export const UPSTREAM_MAP: Record<string, string> = {
  "/auth": AUTH_SERVICE,
};

export function resolveUpstream(path: string): string | null {
  for (const [prefix, url] of Object.entries(UPSTREAM_MAP)) {
    if (path.startsWith(prefix)) return url;
  }
  return null;
}
```

## Task 3: Rate limiter (sliding window in-memory)

- [ ] **Step 1: Criar rate-limiter.ts**

```typescript
// api-gateway/src/middleware/rate-limiter.ts
import { Elysia } from "elysia";

const WINDOW_MS = 60_000;   // 1 minuto
const MAX_REQUESTS = 100;   // 100 req/min por IP

// Map<ip, timestamps[]>
const windows = new Map<string, number[]>();

function isRateLimited(ip: string): boolean {
  const now = Date.now();
  const cutoff = now - WINDOW_MS;
  const timestamps = (windows.get(ip) ?? []).filter((t) => t > cutoff);
  timestamps.push(now);
  windows.set(ip, timestamps);
  return timestamps.length > MAX_REQUESTS;
}

// Limpa IPs inativos a cada minuto para evitar memory leak
setInterval(() => {
  const cutoff = Date.now() - WINDOW_MS;
  for (const [ip, ts] of windows) {
    if (ts[ts.length - 1] < cutoff) windows.delete(ip);
  }
}, WINDOW_MS);

export const rateLimiterPlugin = new Elysia({ name: "rate-limiter" }).derive(
  { as: "global" },
  ({ request, error }) => {
    const ip =
      request.headers.get("x-forwarded-for")?.split(",")[0].trim() ??
      "unknown";
    if (isRateLimited(ip)) {
      throw error(429, { error: "rate limit exceeded" });
    }
    return {};
  }
);
```

## Task 4: Auth guard

**Files:**
- Create: `api-gateway/src/middleware/auth-guard.ts`

- [ ] **Step 1: Criar auth-guard.ts**

```typescript
// api-gateway/src/middleware/auth-guard.ts
import { Elysia } from "elysia";
import { PUBLIC_ROUTES } from "../config/routes";

const AUTH_SERVICE =
  process.env.AUTH_SERVICE_URL ?? "http://auth-service.fintech.svc.cluster.local";

export const authGuardPlugin = new Elysia({ name: "auth-guard" }).derive(
  { as: "global" },
  async ({ request, error }) => {
    const method = request.method;
    const url = new URL(request.url);
    const routeKey = `${method} ${url.pathname}`;

    if (PUBLIC_ROUTES.has(routeKey)) return {};

    const authHeader = request.headers.get("authorization");
    if (!authHeader?.startsWith("Bearer ")) {
      throw error(401, { error: "missing authorization header" });
    }

    const res = await fetch(`${AUTH_SERVICE}/auth/validate`, {
      headers: {
        authorization: authHeader,
        "x-trace-id": request.headers.get("x-trace-id") ?? "",
      },
    }).catch(() => null);

    if (!res || res.status !== 200) {
      throw error(401, { error: "unauthorized" });
    }

    const payload = (await res.json()) as { user_id: string; email: string };
    return { userId: payload.user_id, userEmail: payload.email };
  }
);
```

// api-gateway/src/proxy/proxy.ts
import { resolveUpstream } from "../config/routes";

export async function forwardRequest(
  request: Request,
  traceId: string
): Promise<Response> {
  const url = new URL(request.url);
  const upstream = resolveUpstream(url.pathname);

  if (!upstream) {
    return new Response(JSON.stringify({ error: "not found" }), {
      status: 404,
      headers: { "content-type": "application/json" },
    });
  }

  const targetUrl = upstream + url.pathname + url.search;

  const headers = new Headers(request.headers);
  headers.set("x-trace-id", traceId);
  // Remove hop-by-hop headers
  headers.delete("host");

  const upstreamRes = await fetch(targetUrl, {
    method: request.method,
    headers,
    body:
      request.method !== "GET" && request.method !== "HEAD"
        ? request.body
        : undefined,
  });

  const resHeaders = new Headers(upstreamRes.headers);
  resHeaders.set("x-trace-id", traceId);

  return new Response(upstreamRes.body, {
    status: upstreamRes.status,
    headers: resHeaders,
  });
}
```

// api-gateway/src/index.ts
import { Elysia } from "elysia";
import { tracePlugin } from "./middleware/trace";
import { rateLimiterPlugin } from "./middleware/rate-limiter";
import { authGuardPlugin } from "./middleware/auth-guard";
import { loggerPlugin } from "./middleware/logger";
import { forwardRequest } from "./proxy/proxy";

const app = new Elysia()
  .use(tracePlugin)
  .use(rateLimiterPlugin)
  .use(authGuardPlugin)
  .use(loggerPlugin)
  .get("/healthz", () => ({ status: "ok" }))
  .get("/metrics", () => {
    // Prometheus text format — métricas básicas do processo
    const mem = process.memoryUsage();
    return new Response(
      [
        `# HELP api_gateway_heap_used_bytes Node heap used`,
        `# TYPE api_gateway_heap_used_bytes gauge`,
        `api_gateway_heap_used_bytes ${mem.heapUsed}`,
        `# HELP api_gateway_rss_bytes Process RSS`,
        `# TYPE api_gateway_rss_bytes gauge`,
        `api_gateway_rss_bytes ${mem.rss}`,
      ].join("\n"),
      { headers: { "content-type": "text/plain; version=0.0.4" } }
    );
  })
  // Proxy catch-all: repassa tudo que não é healthz/metrics
  .all("/*", async ({ request, traceId }) =>
    forwardRequest(request, traceId)
  )
  .listen(Number(process.env.PORT ?? 3000));

console.log(
  JSON.stringify({
    level: "info",
    service: "api-gateway",
    msg: `listening on :${app.server?.port}`,
    ts: new Date().toISOString(),
  })
);
```
